mysql> create procedure invo_details(in cid int) begin select CUSTOMER.cust_id,CUSTOMER.name,Invoice_master.inv_id,Invoice_master.Inv_date,PRODUCT.p_name,Invoice_Item.quantity,PRODUCT.price,(Invoice_Item.quantity * PRODUCT.price) as total from CUSTOMER join Invoice_master on CUSTOMER.cust_id=Invoice_master.cust_id join Invoice_Item on Invoice_master.inv_id=Invoice_Item.inv_id join PRODUCT on Invoice_Item.P_id=PRODUCT.P_id where
CUSTOMER.cust_id = cid; end//
