b)DCL Commands
Create two users user1 ,user2
1.GRANT: Give insert and select privilege to user 1 on customer table in the invoice database.
      -> Give update and delete privilege to user 2 on all tables in the invoice database.
2.REVOKE: Remove delete privilege of user 2.

1.mysql> CREATE USER 'dhrisya'@'localhost' IDENTIFIED BY '1234';
Query OK, 0 rows affected (0.01 sec)

mysql> GRANT SELECT ON ksb.EMPLOYEE TO 'dhrisya'@'localhost';
Query OK, 0 rows affected (0.00 sec)

mysql> SHOW GRANTS FOR 'dhrisya'@'localhost';
+-----------------------------------------------------------+
| Grants for dhrisya@localhost                              |
+-----------------------------------------------------------+
| GRANT USAGE ON *.* TO `dhrisya`@`localhost`               |
| GRANT SELECT ON `ksb`.`EMPLOYEE` TO `dhrisya`@`localhost` |
+-----------------------------------------------------------+
2 rows in set (0.00 sec)

mysql> CREATE USER 'user1'@'localhost' IDENTIFIED BY 'passs21';
Query OK, 0 rows affected (0.01 sec)

mysql> CREATE USER 'user2'@'localhost' IDENTIFIED BY 'passs33';
Query OK, 0 rows affected (0.01 sec)

mysql> GRANT INSERT ON INVOICE.CUSTOMER TO 'user1'@'localhost';
Query OK, 0 rows affected (0.00 sec)

mysql> GRANT SELECT ON INVOICE.CUSTOMER TO 'user1'@'localhost';
