1. Create a view for customers who have purchased products.
2. Create a view for invoice products[cname, , inv_id, product_id, quantity]

mysql> create view purchased_customer as select CUSTOMER.cust_id,CUSTOMER.name,PRODUCT.P_name,Invoice_master.Inv_date from CUSTOMER join Invoice_master on CUSTOMER.cust_id=Invoice_master.cust_id join Invoice_Item ON Invoice_master.inv_id = Invoice_Item.inv_id join PRODUCT ON Invoice_Item.P_id = PRODUCT.P_id;
